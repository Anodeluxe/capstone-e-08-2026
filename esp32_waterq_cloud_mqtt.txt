void setValve(int i, bool open) {
  bool changed = (valveOpen[i] != open);
  valveOpen[i] = open;
  digitalWrite(PIN_RELAY[i], open ? LOW : HIGH);   // aktif LOW
  if (changed && mqtt.connected()) {
    publishValveAck(i);
  }
}

void initRelays() {
  for (int i = 0; i < 4; i++) {
    digitalWrite(PIN_RELAY[i], HIGH);  // tulis HIGH DULU supaya relay tidak sempat ON
    pinMode(PIN_RELAY[i], OUTPUT);
    valveOpen[i] = false;
    prevValveOpen[i] = false;
  }
}

// Rata-rata 32 sampel ADC setelah membuang 8 terendah & 8 tertinggi (Volt)
float readAdcVolt(int pin) {
  const int N = 32;
  uint16_t v[N];
  for (int i = 0; i < N; i++) {
    v[i] = analogReadMilliVolts(pin);
    delay(2);
  }
  for (int i = 1; i < N; i++) {            // insertion sort
    uint16_t k = v[i]; int j = i - 1;
    while (j >= 0 && v[j] > k) { v[j + 1] = v[j]; j--; }
    v[j + 1] = k;
  }
  uint32_t sum = 0;
  for (int i = 8; i < N - 8; i++) sum += v[i];
  return (sum / (float)(N - 16)) / 1000.0f;
}

float phFromVolt(float v) {
  float slope = (cal.ph4V - cal.ph7V) / (4.0f - 7.0f); // V per satuan pH (negatif)
  if (fabsf(slope) < 0.01f) slope = -0.18f;            // jaga-jaga kalibrasi rusak
  return 7.0f + (v - cal.ph7V) / slope;
}

// Kurva DFRobot SEN0189 (V pada suplai 5 V), digeser sesuai kalibrasi air jernih
float ntuFromVolt(float v) {
  float va = v + (4.20f - cal.turbClearV);
  if (va >= 4.20f) return 0.0f;
  if (va < 2.50f)  return 3000.0f;
  float ntu = -1120.4f * va * va + 5742.3f * va - 4352.9f;
  return ntu < 0 ? 0 : ntu;
}

// Rumus DFRobot Gravity TDS dengan kompensasi suhu (hasil ppm, faktor 0,5)
float tdsFromVolt(float v, float tempC) {
  float cv = v / (1.0f + 0.02f * (tempC - 25.0f));
  float tds = (133.42f * cv * cv * cv - 255.86f * cv * cv + 857.39f * cv) * 0.5f * cal.tdsK;
  return tds < 0 ? 0 : tds;
}

// Jarak ultrasonik (cm), median 5 pantulan, kecepatan suara dikompensasi suhu
float readDistanceCm(float tempC) {
  float c = 331.3f + 0.606f * tempC;     // m/s
  float d[5]; int n = 0;
  for (int i = 0; i < 5; i++) {
    digitalWrite(PIN_TRIG, LOW);  delayMicroseconds(4);
    digitalWrite(PIN_TRIG, HIGH); delayMicroseconds(20);
    digitalWrite(PIN_TRIG, LOW);
    unsigned long us = pulseIn(PIN_ECHO, HIGH, 30000UL);   // maks ±5 m
    if (us > 0) d[n++] = us * 1e-6f * c * 100.0f / 2.0f;
    delay(60);
  }
  if (n == 0) return NAN;
  for (int i = 1; i < n; i++) {
    float k = d[i]; int j = i - 1;
    while (j >= 0 && d[j] > k) { d[j + 1] = d[j]; j--; }
    d[j + 1] = k;
  }
  return d[n / 2];
}

void nowString(char* buf, size_t len) {
  struct tm t;
  if (staMode && getLocalTime(&t, 0)) {
    strftime(buf, len, "%Y-%m-%d %H:%M:%S", &t);
  } else {
    uint32_t s = millis() / 1000;
    snprintf(buf, len, "uptime %lu:%02lu:%02lu",
             (unsigned long)(s / 3600), (unsigned long)(s / 60 % 60), (unsigned long)(s % 60));
  }
}

// 6. KLASIFIKASI & KEPUTUSAN VALVE
// =====================================================================
bool inLimit(const Limit& L, float ph, float ntu, float tds) {
  return ph >= L.phMin && ph <= L.phMax && ntu <= L.ntuMax && tds <= L.tdsMax;
}

char classify(Reading& r) {
  // Sensor analog dianggap error kalau tegangannya mentok 0 atau mentok atas ADC
  bool phErr   = r.vPh   < 0.05f || r.vPh   > 4.95f || isnan(r.ph);
  bool turbErr = r.vTurb < 0.05f;
  bool tdsErr  = r.vTds  > 3.05f;
  if (phErr || turbErr || tdsErr) {
    snprintf(r.reason, sizeof(r.reason), "Sensor error:%s%s%s",
             phErr ? " pH" : "", turbErr ? " kekeruhan" : "", tdsErr ? " TDS" : "");
    return 'D';
  }
  if (inLimit(LIM_A, r.ph, r.ntu, r.tds)) { strcpy(r.reason, "Layak semua keperluan"); return 'A'; }
  if (inLimit(LIM_B, r.ph, r.ntu, r.tds)) { strcpy(r.reason, "Tidak untuk dapur/konsumsi"); return 'B'; }
  if (inLimit(LIM_C, r.ph, r.ntu, r.tds)) { strcpy(r.reason, "Hanya untuk siram taman"); return 'C'; }
  strcpy(r.reason, "Kualitas buruk, semua ditutup");
  return 'D';
}

void applyValves() {
  bool lowLevel = LOW_LEVEL_CUTOFF && cur.distOk && cur.levelPct < LEVEL_MIN_PCT;
  if (!autoMode) {
    for (int i = 0; i < 4; i++) setValve(i, manualReq[i]);
    return;
  }
  // Konfirmasi kelas (anti cetak-cetek), kelas D langsung berlaku
  if (cur.cls == 'D') {
    appliedCls = 'D'; pendingCls = 'D'; pendingCnt = CONFIRM_COUNT;
  } else if (cur.cls == pendingCls) {
    if (pendingCnt < CONFIRM_COUNT) pendingCnt++;
    if (pendingCnt >= CONFIRM_COUNT) appliedCls = pendingCls;
  } else {
    pendingCls = cur.cls; pendingCnt = 1;
    if (CONFIRM_COUNT <= 1) appliedCls = pendingCls;
  }
  int row = appliedCls - 'A';
  for (int i = 0; i < 4; i++) setValve(i, !lowLevel && VALVE_RULE[row][i]);
}

// =====================================================================
// 7. SAMPLING (dipanggil tiap 10 detik)
// =====================================================================
void takeSample() {
  Reading& r = cur;
  nowString(r.time, sizeof(r.time));

  // Suhu: konversi sudah diminta 10 s lalu (non-blocking), tinggal dibaca
  float t = ds18.getTempCByIndex(0);
  r.tempOk = !(t == DEVICE_DISCONNECTED_C || t < -20.0f || t > 80.0f || t == 85.0f);
  r.tempC  = r.tempOk ? t : NAN;
  if (r.tempOk) lastGoodTemp = t;
  ds18.requestTemperatures();            // minta konversi untuk sampel berikutnya
  float tComp = lastGoodTemp;            // suhu untuk kompensasi TDS & ultrasonik

  r.vPh   = readAdcVolt(PIN_PH)   / DIV_RATIO;
  r.vTurb = readAdcVolt(PIN_TURB) / DIV_RATIO;
  r.vTds  = readAdcVolt(PIN_TDS);

  r.ph  = phFromVolt(r.vPh);
  r.ntu = ntuFromVolt(r.vTurb);
  r.tds = tdsFromVolt(r.vTds, tComp);
  r.ec  = r.tds * 2.0f;                  // µS/cm (TDS = EC x 0,5)

  r.distCm = readDistanceCm(tComp);
  r.distOk = !isnan(r.distCm) && r.distCm > 2.0f && r.distCm < TANK_DEPTH_CM + 30.0f;
  if (r.distOk) {
    float pct = (TANK_DEPTH_CM - r.distCm) / (TANK_DEPTH_CM - FULL_GAP_CM) * 100.0f;
    r.levelPct = constrain(pct, 0.0f, 100.0f);
    r.volumeL  = r.levelPct / 100.0f * TANK_CAPACITY_L;
    lastGoodLevel = r.levelPct;
  } else {
    r.levelPct = NAN; r.volumeL = NAN;
  }

  r.cls = classify(r);
  applyValves();
  sampleCount++;

  hist[histHead] = { millis() / 1000, r.ph, r.ntu, r.tds, r.tempC, r.levelPct };
  histHead = (histHead + 1) % HIST_N;
  if (histCount < HIST_N) histCount++;
}

// 8. FORMAT DATA (CSV, JSON LOKAL, & CLOUD MQTT)
// =====================================================================
String fnum(float v, int dec) {          // NaN -> "null" (JSON)
  if (isnan(v) || isinf(v)) return "null";
  return String(v, dec);
}
String fcsv(float v, int dec) {          // NaN -> kosong (CSV)
  if (isnan(v) || isinf(v)) return "";
  return String(v, dec);
}

void printCsvHeader() {
  Serial.println(F("waktu,ph,kekeruhan_ntu,tds_ppm,ec_uscm,suhu_c,jarak_cm,level_pct,volume_l,"
                   "kelas,v_mandi,v_dapur,v_cuci,v_taman,mode,vph,vturb,vtds"));
}

void printCsvRow() {
  const Reading& r = cur;
  String s = String(r.time) + "," + fcsv(r.ph, 2) + "," + fcsv(r.ntu, 1) + "," +
             fcsv(r.tds, 0) + "," + fcsv(r.ec, 0) + "," + fcsv(r.tempC, 2) + "," +
             fcsv(r.distCm, 1) + "," + fcsv(r.levelPct, 1) + "," + fcsv(r.volumeL, 0) + "," +
             String(r.cls);
  for (int i = 0; i < 4; i++) s += valveOpen[i] ? ",1" : ",0";
  s += autoMode ? ",AUTO" : ",MANUAL";
  s += "," + fcsv(r.vPh, 3) + "," + fcsv(r.vTurb, 3) + "," + fcsv(r.vTds, 3);
  Serial.println(s);
}

  const Reading& r = cur;
  String j;
  j.reserve(900);

  j += ",\"sampel\":" + String(sampleCount);
  j += ",\"interval_s\":" + String(SAMPLE_INTERVAL_MS / 1000);
  j += ",\"ph\":" + fnum(r.ph, 2);
  j += ",\"kekeruhan_ntu\":" + fnum(r.ntu, 1);
  j += ",\"tds_ppm\":" + fnum(r.tds, 0);
  j += ",\"ec_uscm\":" + fnum(r.ec, 0);
  j += ",\"suhu_c\":" + fnum(r.tempC, 2);
  j += ",\"jarak_cm\":" + fnum(r.distCm, 1);
  j += ",\"level_pct\":" + fnum(r.levelPct, 1);
  j += ",\"volume_l\":" + fnum(r.volumeL, 0);
  j += ",\"kelas\":\"" + String(r.cls) + "\"";
  j += ",\"kelas_aktif\":\"" + String(appliedCls) + "\"";
  j += ",\"keterangan\":\"" + String(r.reason) + "\"";
  j += ",\"level_rendah\":" + String((LOW_LEVEL_CUTOFF && r.distOk && r.levelPct < LEVEL_MIN_PCT) ? "true" : "false");
  j += ",\"mode\":\"" + String(autoMode ? "AUTO" : "MANUAL") + "\"";
  j += ",\"valve\":[";
  for (int i = 0; i < 4; i++) {
    if (i) j += ",";
    j += "{\"nama\":\"" + String(VALVE_NAME[i]) + "\",\"buka\":" + (valveOpen[i] ? "true" : "false") + "}";
  }
  j += "],\"tegangan\":{\"ph\":" + fnum(r.vPh, 3) + ",\"turb\":" + fnum(r.vTurb, 3) +
       ",\"tds\":" + fnum(r.vTds, 3) + "}";
  j += ",\"wifi\":\"" + String(staMode ? "STA" : "AP") + "\",\"rssi\":" + String(staMode ? WiFi.RSSI() : 0);
  j += "}";
  return j;
}

String jsonHistory() {
  String j;
  j.reserve(histCount * 60 + 100);
  j = "{\"t\":[";
  String ph = "", ntu = "", tds = "", tmp = "", lvl = "";
  int start = (histHead - histCount + HIST_N) % HIST_N;
  for (int k = 0; k < histCount; k++) {
    const HistPoint& p = hist[(start + k) % HIST_N];
    const char* sep = k ? "," : "";
    j   += sep + String(p.t);
    ph  += sep + fnum(p.ph, 2);
    ntu += sep + fnum(p.ntu, 1);
    tds += sep + fnum(p.tds, 0);
    tmp += sep + fnum(p.temp, 2);
    lvl += sep + fnum(p.level, 1);
  }
  j += "],\"ph\":[" + ph + "],\"ntu\":[" + ntu + "],\"tds\":[" + tds +
       "],\"suhu\":[" + tmp + "],\"level\":[" + lvl + "]}";
  return j;
}

void postToServer() {
  if (strlen(POST_URL) == 0 || WiFi.status() != WL_CONNECTED) return;
  HTTPClient http;
  http.setTimeout(3000);
  if (!http.begin(POST_URL)) return;
  http.addHeader("Content-Type", "application/json");
  int code = http.POST(jsonData());
  if (code <= 0) Serial.printf("# POST gagal: %s\n", http.errorToString(code).c_str());
  http.end();
}

// --- Komunikasi MQTT ke Backend FastAPI (Cloud Broker) ---
void publishValveAck(int idx) {
  if (!mqtt.connected() || idx < 0 || idx >= 4) return;
  char ack[128];
  snprintf(ack, sizeof(ack), "{\"valve_id\":\"%s\",\"is_open\":%s}",
           VALVE_ID_STR[idx], valveOpen[idx] ? "true" : "false");
  mqtt.publish(TOPIC_VALVE_STAT, ack, false);
}

void syncAllValvesToMqtt() {
  for (int i = 0; i < 4; i++) {
    publishValveAck(i);
  }
}

void publishMqttSensor() {
  if (!mqtt.connected()) return;
  const Reading& r = cur;

  // Pastikan nilai valid sesuai skema Pydantic ESP32SensorPayload di backend
  float safePh    = isnan(r.ph)       ? 7.0f          : constrain(r.ph, 0.0f, 14.0f);
  float safeTurb  = isnan(r.ntu)      ? 0.0f          : max(0.0f, r.ntu);
  float safeTds   = isnan(r.tds)      ? 0.0f          : max(0.0f, r.tds);
  float safeTemp  = isnan(r.tempC)    ? constrain(lastGoodTemp, 0.0f, 100.0f) : constrain(r.tempC, 0.0f, 100.0f);
  float safeLevel = isnan(r.levelPct) ? constrain(lastGoodLevel, 0.0f, 100.0f) : constrain(r.levelPct, 0.0f, 100.0f);

  char payload[256];
  snprintf(payload, sizeof(payload),
           "{\"device_id\":\"%s\",\"timestamp\":0,\"ph\":%.2f,\"turbidity\":%.2f,"
           "\"tds\":%.2f,\"temperature\":%.2f,\"water_level\":%.2f}",
           DEVICE_ID, safePh, safeTurb, safeTds, safeTemp, safeLevel);

  if (mqtt.publish(TOPIC_SENSORS, payload, false)) {
    Serial.printf("# [MQTT] Terkirim ke %s: %s\n", TOPIC_SENSORS, payload);
  } else {
    Serial.println(F("# [MQTT] Gagal publish data sensor"));
  }
}

void onMqttMessage(char* topic, byte* payload, unsigned int length) {
  String msg;
  msg.reserve(length + 1);
  for (unsigned int i = 0; i < length; i++) msg += (char)payload[i];

  if (String(topic) != TOPIC_VALVE_CMD) return;
  Serial.printf("# [MQTT] Perintah valve diterima: %s\n", msg.c_str());

  // Identifikasi valve_id: bathroom, kitchen, laundry, garden
  int idx = -1;
  for (int i = 0; i < 4; i++) {
    String pattern = String("\"valve_id\":\"") + VALVE_ID_STR[i] + "\"";
    String patternSp = String("\"valve_id\": \"") + VALVE_ID_STR[i] + "\"";
    if (msg.indexOf(pattern) >= 0 || msg.indexOf(patternSp) >= 0) {
      idx = i;
      break;
    }
  }
  if (idx < 0) return;

  bool isOpen = (msg.indexOf("\"action\":\"open\"") >= 0 || msg.indexOf("\"action\": \"open\"") >= 0);
  bool isSystemCmd = (msg.indexOf("\"triggered_by\":\"system\"") >= 0 || msg.indexOf("\"triggered_by\": \"system\"") >= 0);

  // Jika perintah berasal dari tombol manual di Web App Dashboard (bukan auto system),
  // pindahkan ESP32 ke mode MANUAL agar tidak langsung ditimpa oleh loop lokal.
  if (!isSystemCmd) {
    if (autoMode) {
      autoMode = false;
      for (int i = 0; i < 4; i++) manualReq[i] = valveOpen[i];
      Serial.println(F("# [MQTT] Override manual dari Web App -> pindah ke mode MANUAL"));
    }
    manualReq[idx] = isOpen;
    applyValves();
  } else if (autoMode) {
    setValve(idx, isOpen);
  }

  // Kirim konfirmasi status terbaru kembali ke backend
  publishValveAck(idx);
}

void ensureMqttConnected() {
  if (!staMode || WiFi.status() != WL_CONNECTED) return;
  if (mqtt.connected()) return;

  uint32_t now = millis();
  if (now - lastMqttRetryMs < 5000) return;
  lastMqttRetryMs = now;

  String clientId = String("esp32_waterq_e08_") + String((uint32_t)ESP.getEfuseMac(), HEX);
  Serial.printf("# [MQTT] Menyambung ke %s:%d sebagai %s...", MQTT_BROKER, MQTT_PORT, clientId.c_str());
  if (mqtt.connect(clientId.c_str())) {
    Serial.println(F(" OK!"));
    mqtt.subscribe(TOPIC_VALVE_CMD);
    syncAllValvesToMqtt();
  } else {
    Serial.printf(" gagal (rc=%d), coba lagi 5 detik\n", mqtt.state());
  }
}

void sendJson(int code, const String& body) {
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.sendHeader("Cache-Control", "no-store");
  server.send(code, "application/json", body);
}

void setupWeb() {
  server.on("/", HTTP_GET, []() { server.send(200, "text/html", INDEX_HTML); });
  server.on("/api/data", HTTP_GET, []() { sendJson(200, jsonData()); });
  server.on("/api/history", HTTP_GET, []() { sendJson(200, jsonHistory()); });

  // POST /api/mode?auto=1  (AUTO) atau auto=0 (MANUAL)
  server.on("/api/mode", HTTP_POST, []() {
    if (!server.hasArg("auto")) { sendJson(400, "{\"error\":\"butuh parameter auto\"}"); return; }
    autoMode = server.arg("auto") == "1";
    if (!autoMode) for (int i = 0; i < 4; i++) manualReq[i] = valveOpen[i]; // mulai dari kondisi sekarang
    applyValves();
    sendJson(200, jsonData());
  });

  // POST /api/valve?ch=1..4&state=0/1  (hanya di mode MANUAL)
  server.on("/api/valve", HTTP_POST, []() {
    if (autoMode) { sendJson(409, "{\"error\":\"pindah ke mode MANUAL dulu\"}"); return; }
    int ch = server.arg("ch").toInt();
    if (ch < 1 || ch > 4 || !server.hasArg("state")) { sendJson(400, "{\"error\":\"ch 1-4, state 0/1\"}"); return; }
    manualReq[ch - 1] = server.arg("state") == "1";
    applyValves();
    sendJson(200, jsonData());
  });

  server.onNotFound([]() { server.send(404, "text/plain", "404"); });
  server.begin();
}

void setupWiFi() {
  WiFi.setHostname(HOSTNAME);           // harus sebelum WiFi.mode()/begin()
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  Serial.printf("# Menyambung ke WiFi \"%s\"", WIFI_SSID);
  uint32_t t0 = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - t0 < 15000) {
    delay(500); Serial.print('.');
  }
  Serial.println();
  if (WiFi.status() == WL_CONNECTED) {
    staMode = true;
    WiFi.setAutoReconnect(true);
    configTime(GMT_OFFSET_SEC, 0, "pool.ntp.org", "time.google.com");
    Serial.printf("# WiFi OK, IP: %s  ->  http://%s.local\n",
                  WiFi.localIP().toString().c_str(), HOSTNAME);
  } else {
    staMode = false;
    WiFi.disconnect(true);
    WiFi.mode(WIFI_AP);
    WiFi.softAP(AP_SSID, AP_PASS);
    Serial.printf("# WiFi gagal -> hotspot \"%s\" (pass %s), buka http://%s\n",
                  AP_SSID, AP_PASS, WiFi.softAPIP().toString().c_str());
  }
  if (MDNS.begin(HOSTNAME)) MDNS.addService("http", "tcp", 80);

  // Setup Cloud MQTT Broker
  mqtt.setServer(MQTT_BROKER, MQTT_PORT);
  mqtt.setCallback(onMqttMessage);
  mqtt.setBufferSize(512);
}

void loadCal() {
  prefs.begin("waterq", true);
  cal.ph7V       = prefs.getFloat("ph7V", 2.50f);
  cal.ph4V       = prefs.getFloat("ph4V", 3.05f);
  cal.turbClearV = prefs.getFloat("turbV", 4.20f);
  cal.tdsK       = prefs.getFloat("tdsK", 1.00f);
  prefs.end();
}

void saveCal() {
  prefs.begin("waterq", false);
  prefs.putFloat("ph7V", cal.ph7V);
  prefs.putFloat("ph4V", cal.ph4V);
  prefs.putFloat("turbV", cal.turbClearV);
  prefs.putFloat("tdsK", cal.tdsK);
  prefs.end();
}

void showCal() {
  Serial.printf("# Kalibrasi: pH7=%.3f V  pH4=%.3f V  (slope %.4f V/pH)  turbJernih=%.3f V  tdsK=%.3f\n",
                cal.ph7V, cal.ph4V, (cal.ph4V - cal.ph7V) / -3.0f, cal.turbClearV, cal.tdsK);
}

// Rata-rata 10 x readAdcVolt (±1 detik) untuk kalibrasi yang stabil
float avgVolt(int pin, float div) {
  float s = 0;
  for (int i = 0; i < 10; i++) s += readAdcVolt(pin);
  return s / 10.0f / div;
}

void printHelp() {
  Serial.println(F(
    "# ---- Perintah (ketik di Serial Monitor, akhiri Enter) ----\n"
    "# help              daftar perintah\n"
    "# show              tampilkan kalibrasi\n"
    "# raw               tegangan sensor saat ini\n"
    "# cal7              probe pH di buffer pH 7,00 (tunggu 1-2 menit stabil)\n"
    "# cal4              probe pH di buffer pH 4,00\n"
    "# calturb           sensor kekeruhan di air jernih (0 NTU)\n"
    "# caltds <ppm>      sensor TDS di larutan standar, mis. caltds 707\n"
    "# reset             kembalikan kalibrasi default\n"
    "# auto | manual     mode valve\n"
    "# valve <1-4> <0/1> buka(1)/tutup(0) valve (mode manual)\n"
    "# now               ambil sampel sekarang\n"
    "# header            cetak ulang header CSV"));
}

void handleCommand(String cmd) {
  cmd.trim();
  String lc = cmd; lc.toLowerCase();
  if (lc.length() == 0) return;

  if (lc == "help") printHelp();
  else if (lc == "show") showCal();
  else if (lc == "raw") {
    Serial.printf("# Vph=%.3f V  Vturb=%.3f V  Vtds=%.3f V\n",
                  readAdcVolt(PIN_PH) / DIV_RATIO, readAdcVolt(PIN_TURB) / DIV_RATIO, readAdcVolt(PIN_TDS));
  }
  else if (lc == "cal7") { cal.ph7V = avgVolt(PIN_PH, DIV_RATIO); saveCal(); showCal(); }
  else if (lc == "cal4") { cal.ph4V = avgVolt(PIN_PH, DIV_RATIO); saveCal(); showCal(); }
  else if (lc == "calturb") { cal.turbClearV = avgVolt(PIN_TURB, DIV_RATIO); saveCal(); showCal(); }
  else if (lc.startsWith("caltds")) {
    float ref = lc.substring(6).toFloat();
    if (ref <= 0) { Serial.println("# format: caltds 707"); return; }
    float v = avgVolt(PIN_TDS, 1.0f);
    float oldK = cal.tdsK; cal.tdsK = 1.0f;
    float raw = tdsFromVolt(v, lastGoodTemp);
    if (raw < 1.0f) { cal.tdsK = oldK; Serial.println("# TDS terlalu kecil, cek probe"); return; }
    cal.tdsK = ref / raw; saveCal(); showCal();
  }
  else if (lc == "reset") {
    cal = {2.50f, 3.05f, 4.20f, 1.00f}; saveCal(); showCal();
  }
  else if (lc == "auto")   { autoMode = true;  applyValves(); Serial.println("# mode AUTO"); }
  else if (lc == "manual") {
    autoMode = false; for (int i = 0; i < 4; i++) manualReq[i] = valveOpen[i];
    applyValves(); Serial.println("# mode MANUAL");
  }
  else if (lc.startsWith("valve")) {
    int ch = -1, st = -1;
    sscanf(lc.c_str(), "valve %d %d", &ch, &st);
    if (autoMode) Serial.println("# ketik 'manual' dulu");
    else if (ch < 1 || ch > 4 || (st != 0 && st != 1)) Serial.println("# format: valve 1 1");
    else { manualReq[ch - 1] = st; applyValves(); Serial.printf("# %s %s\n", VALVE_NAME[ch - 1], st ? "BUKA" : "TUTUP"); }
  }
  else if (lc == "now") { takeSample(); printCsvRow(); publishMqttSensor(); }
  else if (lc == "header") printCsvHeader();
  else Serial.println("# perintah tidak dikenal, ketik help");
}

void pollSerial() {
  while (Serial.available()) {
    char c = Serial.read();
    if (c == '\n' || c == '\r') { if (serialLine.length()) { handleCommand(serialLine); serialLine = ""; } }
    else if (serialLine.length() < 64) serialLine += c;
  }
}

void setup() {
  initRelays();                         // PALING AWAL: semua valve tertutup
  Serial.begin(115200);
  delay(200);
  Serial.println(F("\n# ===== WaterQ E08 — ESP32 38 pin (Cloud MQTT) ====="));

  pinMode(PIN_TRIG, OUTPUT);
  digitalWrite(PIN_TRIG, LOW);
  pinMode(PIN_ECHO, INPUT);
  analogReadResolution(12);             // atenuasi default 11 dB (0–3,1 V)

  loadCal();
  showCal();

  ds18.begin();
  ds18.setResolution(12);
  ds18.setWaitForConversion(false);     // non-blocking
  ds18.requestTemperatures();
  Serial.printf("# DS18B20 terdeteksi: %d\n", ds18.getDeviceCount());

  setupWiFi();
  setupWeb();
  ensureMqttConnected();
  printHelp();

  delay(800);                           // tunggu konversi suhu pertama (750 ms)
  printCsvHeader();
  takeSample();
  printCsvRow();
  publishMqttSensor();
  postToServer();
  lastSampleMs = millis();
}

void loop() {
  server.handleClient();
  pollSerial();

  if (staMode && WiFi.status() == WL_CONNECTED) {
    if (!mqtt.connected()) ensureMqttConnected();
    mqtt.loop();
  }

  uint32_t now = millis();
  if (now - lastSampleMs >= SAMPLE_INTERVAL_MS) {
    lastSampleMs += SAMPLE_INTERVAL_MS;
    if (now - lastSampleMs >= SAMPLE_INTERVAL_MS) lastSampleMs = now;  // kalau sempat tertinggal jauh
    takeSample();
    printCsvRow();
    publishMqttSensor();
    postToServer();
  }

  // Coba sambung ulang WiFi tiap 30 s kalau putus (mode STA)
  static uint32_t lastWifiCheck = 0;
  if (staMode && now - lastWifiCheck > 30000) {
    lastWifiCheck = now;
    if (WiFi.status() != WL_CONNECTED) WiFi.reconnect();
  }
}
